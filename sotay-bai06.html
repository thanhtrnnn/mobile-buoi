data class Category(
    var id: Int = 0,
    var name: String = "",
    var icon: String = Icons.DEFAULT,
    var note: String = "",
    var type: CategoryType = CategoryType(),
    var parent: Category? = null
) : Serializable {

    /** Mục cha ở bậc 0, con bậc 1, cháu bậc 2 — CategoryAdapter dùng để thụt lề. */
    fun level(): Int {
        var level = 0
        var p = parent
        while (p != null) { level++; p = p.parent }
        return level
    }

Bản đầu gieo bộ mục theo slide 7 (Sinh hoạt › Ăn uống › Ăn nhà hàng…). Khi đổi sang bộ 23 mục hiện tại, DB_VERSION được tăng lên 2 để máy đã cài bản cũ chạy onUpgrade(): xóa ba bảng rồi gọi lại onCreate(). Xóa theo thứ tự giao dịch → mục → kiểu.

loadCategories — nối con về đúng đối tượng cha

Đọc bảng xong mới có đủ các đối tượng, nên phải làm hai lượt: lượt 1 tạo mọi Category và nhớ idParent của từng cái; lượt 2 gán parent bằng cách tra id trong map.

WalletDAO.ktloadCategories (rút gọn)
val all = LinkedHashMap<Int, Category>()
val idParentOf = HashMap<Int, Int>()
while (cursor.moveToNext()) {
    val category = categoryOf(cursor, type)
    all[category.id] = category
    if (!cursor.isNull(indexParent)) idParentOf[category.id] = cursor.getInt(indexParent)
}
cursor.close()

// Lượt 2: lúc này mọi mục đã có trong map, gán cha được rồi
for ((id, idParent) in idParentOf) all[id]?.parent = all[idParent]
Vì sao không gán cha ngay trong vòng while? Cha có thể nằm ở dòng phía sau con (ví dụ sửa một mục cũ thành con của một mục mới tạo). Lúc đọc tới con thì cha chưa được tạo ra.

WalletDAO.ktgetCategories
val all = loadCategories(idType)      // 1. mọi mục của kiểu, đã nối cha
val usage = countUsage()             // 2. idCategory -> số giao dịch (GROUP BY)

// 3. cộng dồn lên cha: mỗi mục đi ngược lên hết đường cha, cộng số lần dùng của nó vào từng bậc
val branch = HashMap<Int, Int>()
for (category in all.values) {
    val own = usage[category.id] ?: 0
    if (own == 0) continue
    var node: Category? = category
    while (node != null) {
        branch[node.id] = (branch[node.id] ?: 0) + own
        node = node.parent
    }
}

// 4. tách gốc và con, sắp xếp, rồi duỗi cây thành danh sách
val collator = Collator.getInstance(Locale("vi", "VN"))

roots.sortWith(compareByDescending<Category> { branch[it.id] ?: 0 }.then(byName))

Ví dụ tính tay

Giả sử có cây: Ăn uống › Ăn nhà hàng, và các giao dịch: Ăn nhà hàng ×2, Giải trí ×1, Ăn uống ×1.

MụcTự dùng (usage)Cả nhánh (branch)
Ăn uống11 + 2 = 3
  Ăn nhà hàng22
Giải trí11
Các mục khác00 → xếp abc với nhau

Kết quả: Ăn uống (3), Ăn nhà hàng ngay dưới, rồi Giải trí (1), rồi các mục còn lại theo abc.

Vì sao cần Collator? So chuỗi thô là so mã Unicode: chữ Ă (U+0102) lớn hơn T nên "Ăn uống" bị xếp sau "Thuê nhà". Collator tiếng Việt xếp đúng A → Ă → Â → B … D → Đ → E. Vì thế trong danh sách thu, "Đầu tư" đứng trước "Khác".

06Hiện số tiền k / M / B (slide 11)

Money.kt
private val UNITS = listOf("", "k", "M", "B")

fun format(amount: Float): String {
    var value = amount
    var unit = 0
    while (value >= 1000f && unit < UNITS.size - 1) {   // chia 1000 tới khi dưới 1000
        value /= 1000f
        unit++
    }
    // Làm tròn mà chạm 1000 thì phải lên đơn vị kế: 999999 hiện "1M"
    if (unit in 1 until UNITS.size - 1 && Math.round(value * 100) >= 100_000) {
        value /= 1000f
        unit++
    }
    return short(value) + UNITS[unit]
}

/** Làm tròn 2 chữ số, dùng dấu phẩy thập phân và bỏ phần ",00" thừa. */
private fun short(value: Float): String {
    val text = String.format(Locale.US, "%.2f", value)
        .trimEnd('0')
        .trimEnd('.')
    return text.replace('.', ',')
}
  • Vì sao không dùng when so 1.000 / 1.000.000 như cách nghĩ đầu tiên? Vì phải chọn đơn vị sau khi làm tròn. So trước thì 999999 < 1.000.000 nên chọn k, chia ra 999,999, làm tròn thành "1000k".
  • unit in 1 until 3: số dưới 1000 không bị đẩy lên k (999 vẫn là 999), và B là đơn vị cuối, không có gì để lên nữa.

Kết quả chạy thật (chạy Money.format bằng kotlinc trên máy tính):

Số tiềnHiệnGhi chú

10001k"1,00" bị cắt thành "1"
15001,5k"1,50" bị cắt số 0 cuối
999499999,5k
15370001,54MĐúng ví dụ slide 11

  • Vì sao Locale.US rồi mới đổi dấu? Để chắc chắn %.2f ra dấu chấm, rồi mới tự thay bằng dấu phẩy. Nếu để máy tự chọn locale thì có máy ra phẩy sẵn, có máy ra chấm, và trimEnd('.') sẽ không cắt được.
  • Vì sao cắt số 0? Để "1,00k" thành "1k", "1,50k" thành "1,5k" cho gọn. Nếu thầy muốn luôn đủ 2 chữ số thì bỏ hai dòng trimEnd.
  • Tổng thu/chi ở đầu trang dùng cùng hàm này, nên cũng ra k/M/B.

Nút on/off thu/chi

Slide ghi "thu/chi - làm nút on/off". Dùng Switch đặt giữa hai chữ Chi và Thu: tắt = Chi, bật = Thu.

swType.isChecked = idType == CategoryType.ID_THU

Spinner có tam giác và rộng bằng các ô khác

  • Nền Spinner là bg_spinner.xml: một layer-list gồm khung bg_input + tam giác ic_dropdown neo mép phải (gravity="end|center_vertical"). paddingEnd="50dp" để chữ không đè lên tam giác.

CategoryAdapter.kt
override fun getView(...)         = bind(items[position], ..., indent = false)  // ô đang chọn

// Mỗi bậc thụt vào 40dp, cộng với 15dp lề sẵn có của dòng

Logo lưu thế nào

Cột icon chỉ lưu một tên ngắn như "anuong", "xe". Icons.resOf(name) đổi tên đó sang R.drawable.ic_anuong… Không lưu thẳng R.drawable.xxx vào CSDL vì số đó do aapt2 sinh ra và có thể đổi mỗi lần build. LogoAdapter hiện đủ 22 logo, mỗi dòng có hình bên cạnh tên.

Logo tô xanh / đỏ theo thu / chi

File vector vẽ nét đen; màu được tô lúc chạy bằng imgIcon.setColorFilter(màu), nên một logo dùng được cho cả hai kiểu. Màu lấy từ Icons.colorOf(context, idType) (R.color.thu / R.color.chi). Tô ở ba chỗ:

  • ShowTransFrag.bind — dòng ở Home, cùng màu với tên và số tiền.
  • CategoryAdapter — theo kiểu của từng mục (c.type.id).
  • LogoAdapter — theo kiểu đang gạt trên màn Thêm/Sửa mục. Gạt Switch thì gán logoAdapter.idType rồi notifyDataSetChanged() để vẽ lại.
Vì sao colorFilter không tô luôn khung viền? Khung là background (bg_list_item), còn colorFilter chỉ áp lên ảnh chính (src) của ImageView.

K01Vì sao 999.999 từng hiện "1000k", sửa thế nào (đã sửa)›

Code hiện đã đúng (xem mục 06). Kịch bản này là để giải thích lỗi của bản đầu và cách nghĩ khi sửa: bản đầu chọn đơn vị trước khi làm tròn, 999999 < 1.000.000 nên chọn k, chia ra 999,999, làm tròn 2 số thành 1000 → "1000k". Phải chọn lại đơn vị sau khi làm tròn — đoạn dưới chính là code đang chạy.

Money.kt
Money.ktthay hàm format
private val UNITS = listOf("", "k", "M", "B")

fun format(amount: Float): String {
    var value = amount
    var unit = 0
    while (value >= 1000f && unit < UNITS.size - 1) {
        value /= 1000f
        unit++
    }
    // 999999 chia 1000 được 999,999; làm tròn 2 số thành 1000 -> lên đơn vị kế
    if (unit in 1 until UNITS.size - 1 && Math.round(value * 100) >= 100_000) {
        value /= 1000f
        unit++
    }
    return short(value) + UNITS[unit]
}

Đã chạy thử: 999 → 999, 1000 → 1k, 999499 → 999,5k, 999999 → 1M, 1537000 → 1,54M. Điều kiện unit in 1 until 3 để số dưới 1000 không bị đẩy lên k, và B là đơn vị cuối không có gì để lên nữa.

Money.kt
fun format(amount: Float): String = when {
    amount >= 1_000_000_000f -> short(amount / 1_000_000_000f) + "B"
    amount >= 1_000_000f -> short(amount / 1_000_000f) + "M"
    amount >= 1_000f -> short(amount / 1_000f) + "k"
    else -> String.format(Locale.US, "%.0f", amount)
}

private fun short(value: Float): String =
    String.format(Locale.US, "%.2f", value).replace('.', ',')

K09Hiện giao dịch cả tháng thay vì một ngày›

Ngày lưu dạng dd/MM/yyyy luôn đủ 10 ký tự, nên "mọi ngày của tháng 9/2026" là mọi chuỗi kết thúc bằng /09/2026 — lọc được bằng LIKE.

fun getTransactionsOfMonth(month: Int, year: Int): ArrayList<Transaction> {
    val suffix = String.format(Locale.US, "%%/%02d/%04d", month, year)   // "%/09/2026"
    val db = dbHelper.readableDatabase
    val cursor = db.rawQuery(
        "SELECT ${DBHelper.COL_DATE} FROM ${DBHelper.TB_TRANSACTION} " +
            "WHERE ${DBHelper.COL_DATE} LIKE ? GROUP BY ${DBHelper.COL_DATE}",
        arrayOf(suffix)
    )
    val list = ArrayList<Transaction>()
    while (cursor.moveToNext()) list.addAll(getTransactions(toDate(cursor.getString(0))))
    cursor.close()
    return list
}

Tìm các ngày có giao dịch trong tháng rồi dùng lại getTransactions cho từng ngày, khỏi viết lại phần đọc Cursor. Ở HomeAct.reload() đổi dao.getTransactions(today) thành dao.getTransactionsOfMonth(tháng, năm); phần cộng tổng giữ nguyên.

private var maxIdBefore = -1

maxIdBefore = categories.maxOfOrNull { it?.id ?: 0 } ?: 0

override fun onResume() {
    super.onResume()
    var keep = selectedCategory()?.id ?: 0
    if (maxIdBefore >= 0) {
        val newest = dao.getCategories(idType).maxOfOrNull { it.id } ?: 0
        if (newest > maxIdBefore) keep = newest
        maxIdBefore = -1
    }

private fun countUsage(): HashMap<Int, Int> {
    // Mốc 30 ngày trước. Không lọc được bằng SQL vì cột date là chuỗi
    // "dd/MM/yyyy": so chuỗi thì "05/10/2026" < "20/09/2026" là sai
    val from = Calendar.getInstance()
    from.add(Calendar.DAY_OF_MONTH, -30)

    val db = dbHelper.readableDatabase
    val cursor = db.rawQuery(
        "SELECT ${DBHelper.COL_ID_CATEGORY}, ${DBHelper.COL_DATE} FROM ${DBHelper.TB_TRANSACTION}",
        null
    )
    val usage = HashMap<Int, Int>()
    while (cursor.moveToNext()) {
        if (toDate(cursor.getString(1)).before(from.time)) continue
        val id = cursor.getInt(0)
        usage[id] = (usage[id] ?: 0) + 1
    }
    cursor.close()
    return usage
}

Nhớ thêm import java.util.Calendar. Nếu ngày được lưu dạng yyyy-MM-dd (K22) thì lọc thẳng bằng WHERE date >= ? trong SQL được.

/** id của mục [id] và mọi con cháu của nó. */
private fun branchIds(id: Int): ArrayList<Int> {
    val db = dbHelper.readableDatabase
    val branch = ArrayList<Int>()
    var frontier = listOf(id)
    while (frontier.isNotEmpty()) {
        branch.addAll(frontier)
        val next = ArrayList<Int>()
        for (parentId in frontier) {
            val cursor = db.rawQuery(
                "SELECT ${DBHelper.COL_ID} FROM ${DBHelper.TB_CATEGORY} WHERE ${DBHelper.COL_ID_PARENT} = ?",
                arrayOf(parentId.toString())
            )
            while (cursor.moveToNext()) next.add(cursor.getInt(0))
            cursor.close()
        }
        frontier = next
    }
    return branch
}

toText/toDate đang dùng chung cho cả giao diện và CSDL, nên không đổi thẳng mẫu được (màn hình phải vẫn hiện dd/MM/yyyy). Thêm cặp hàm riêng cho CSDL:

WalletDAO.ktcompanion + chỗ đọc/ghi
private fun dbFormat() = SimpleDateFormat("yyyy-MM-dd", Locale.US)
fun toDbText(date: Date): String = dbFormat().format(date)
fun fromDbText(text: String): Date = dbFormat().parse(text) ?: Date()

// valuesOf(t):       put(COL_DATE, toDbText(t.date))
// getTransactions:   WHERE date = ?  với arrayOf(toDbText(date))
//                    date = fromDbText(cursor.getString(...))
// lọc khoảng:        WHERE date BETWEEN ? AND ?

Dạng năm-tháng-ngày so chuỗi đúng bằng so ngày ("2026-09-20" < "2026-10-05"), nên BETWEEN, ORDER BY date đều đúng. Tăng DB_VERSION vì dữ liệu cũ theo mẫu khác.

Xem file db — phải build bản debuggable

Đã thử trên máy ảo Pixel_Tablet: đây là bản production nên adb root bị từ chối, trên máy cũng không có sqlite3, và run-as báo package not debuggable vì script không bật cờ debug. Cách làm được: thêm --debug-mode vào lệnh aapt2 link (dòng --auto-add-overlay), build lại, rồi kéo file về máy tính:

13Lỗi đã biết

Biết trước để tự nói ra khi bị hỏi, và biết cách sửa (số K ở cuối là kịch bản sửa tương ứng).

1. (Đã sửa) 999.999 hiện "1000k"

Bản đầu chọn đơn vị trước khi làm tròn nên 999999 ra "1000k", lệch slide 11. Đã sửa Money.format để chọn lại đơn vị sau khi làm tròn; chạy thử ra "1M". Xem mục 06 và K01.

2. Số tiền kiểu Float mất chính xác với số lớn

Float chỉ giữ khoảng 7 chữ số có nghĩa. Chạy thử: 123456789f thành 123456792. Slide 9 quy định amount : float nên giữ đúng slide; làm thật thì dùng Long (tiền VNĐ không có phần lẻ) hoặc Double, cột SQLite vẫn là số.

3. Đổi kiểu của một mục có con thì con bị "mồ côi"

editCategory chỉ sửa đúng dòng đó. Con cháu vẫn giữ idType cũ, mà loadCategories(kiểu cũ) không còn thấy cha (cha đã sang kiểu kia), nên parent thành null và con hiện ra như một mục cha trong danh sách kiểu cũ. Sửa ở K18. (Kết luận từ đọc code; dữ liệu mẫu không có mục con nên phải tự tạo mới thấy.)

4. Ngày lưu dạng dd/MM/yyyy không so sánh được bằng SQL

So chuỗi thì "05/10/2026" < "20/09/2026" — sai. Lọc theo đúng một ngày (=) hoặc theo tháng (LIKE '%/09/2026') thì vẫn được, nhưng lọc khoảng và ORDER BY date thì không. Xem K13, K22.

5. Để app mở qua nửa đêm, Home vẫn hiện ngày hôm trước

today được gán một lần lúc tạo HomeAct. Sửa: trong onResume() gán lại today = Date() và lblDate.text trước khi reload() (nếu đã làm K04 thì chỉ làm khi người dùng chưa chọn ngày khác).

6. WalletDAO.total() không ai gọi

HomeAct.reload() tự cộng trên danh sách vừa đọc, khỏi hỏi CSDL thêm hai lần. total() còn lại là hàm thừa; bỏ đi hoặc dùng nó thay cho vòng cộng đều được.

7. Xóa mục là xóa luôn giao dịch

Cố ý, để không còn giao dịch trỏ vào mục không tồn tại; dialog có báo trước. Nhưng dễ mất dữ liệu nếu bấm nhầm. Cách an toàn hơn ở K17.

8. Cách vào màn Sửa mục khó đoán

Phải bấm giữ Spinner mục ở màn giao dịch. Khi demo phải chỉ rõ. Không dùng bấm thường vì bấm thường là mở danh sách; và Spinner không cho setOnClickListener (ném lỗi Don't call setOnClickListener for an AdapterView).

9. Tăng DB_VERSION là mất giao dịch

onUpgrade xóa ba bảng rồi tạo lại. Lần đổi bộ mục mẫu (lên version 2) đã xóa giao dịch trên máy thử.

14Bộ câu hỏi & trả lời

A · CSDL và quan hệ

QSao phải có bảng tblType, không dùng một cột true/false "là thu"?›

Slide 10 thiết kế vậy, và nó mở rộng được: muốn thêm kiểu thứ ba (ví dụ "chuyển tiền giữa hai ví") chỉ cần thêm một dòng vào tblType, không phải đổi cấu trúc bảng. Cột boolean chỉ chứa được hai kiểu.

QidParent bằng NULL nghĩa là gì? Sao không để 0?›

NULL = không có cha, tức là mục cha ở bậc 0. Để 0 thì trông như trỏ vào mục có id 0 (không tồn tại), dễ nhầm. Code đọc bằng cursor.isNull(...), ghi bằng putNull(COL_ID_PARENT).

QCó khai báo FOREIGN KEY không? Ai đảm bảo idCategory hợp lệ?›

Không khai báo. SQLite trên Android mặc định cũng không bật kiểm tra khóa ngoại (phải gọi db.setForeignKeyConstraintsEnabled(true)). Tính đúng do code tự lo: giao dịch chỉ tạo được từ mục chọn trong Spinner, và deleteCategory tự xóa giao dịch của cả nhánh trước khi xóa mục.

QTổng thu/chi đang tính ở đâu? Sao không dùng SUM trong SQL?›

Tính trong HomeAct.reload() bằng vòng cộng trên danh sách vừa đọc. Danh sách đã có trong tay nên cộng luôn, khỏi hỏi CSDL thêm. Dùng SQL thì phải JOIN với tblCategory vì kiểu nằm ở bảng mục: SELECT SUM(t.amount) FROM tblTransaction t JOIN tblCategory c ON c.id = t.idCategory WHERE t.date = ? AND c.idType = ?.

QVì sao một WalletDAO chứ không tách TransactionDAO, CategoryDAO?›

Bài nhỏ, ba bảng dính nhau (đọc giao dịch cần nạp cây mục, xếp mục cần đếm giao dịch). Tách được: CategoryDAO lo tblType + tblCategory, TransactionDAO lo tblTransaction và gọi CategoryDAO để lấy mục. Cả hai dùng chung một DBHelper.

B · Thuật toán xếp mục

Q"Lần đầu xếp abc" được code ở đâu?›

Không có nhánh riêng. Lần đầu chưa có giao dịch nên mọi branch bằng 0; compareByDescending { branch } coi tất cả bằng nhau và .then(byName) quyết định — ra đúng abc.

Q"Tổng lần sử dụng của cả nhánh" tính thế nào?›

Với mỗi mục có số lần dùng own > 0, đi từ nó ngược lên theo parent tới gốc, cộng own vào từng mục trên đường. Xong thì mỗi mục chứa tổng của chính nó và mọi con cháu. Mỗi mục đi tối đa bằng độ sâu cây (≤ 3 bước) nên rất nhanh.

QVì sao danh sách trả về đã "duỗi phẳng" chứ không phải cây?›

Spinner chỉ hiện được một danh sách phẳng. Duỗi theo thứ tự cha → con → cháu (duyệt theo chiều sâu, appendBranch đệ quy), rồi CategoryAdapter dùng level() để thụt lề là nhìn ra cây.

QHai mục cha cùng số lần dùng thì sao?›

Xếp theo abc giữa hai mục đó, nhờ .then(byName).

QCollator để làm gì? Bỏ đi thì sao?›

So chuỗi theo luật tiếng Việt. Bỏ đi thì so theo mã Unicode: "Ăn uống" (Ă = U+0102) bị xếp sau mọi chữ Latin thường, "Đầu tư" (Đ = U+0110) cũng vậy. Đã gặp thật ở bản đầu: "Thuê nhà" đứng trước "Ăn uống".

C · Spinner, adapter, giao diện

QgetView với getDropDownView khác nhau thế nào?›

getView vẽ ô đang chọn khi Spinner đóng; getDropDownView vẽ từng dòng trong danh sách xổ ra. Bài này cho dòng xổ ra thụt lề theo bậc, còn ô đang chọn thì không.

QSao phải tách "đổi kiểu" ra listener, và gán isChecked trước?›

Slide 12 muốn đổi kiểu là mục tự cập nhật, nên listener gọi reloadCategories(0). Nếu gắn listener trước rồi mới gán isChecked theo giao dịch đang sửa, chính lần gán đó chạy listener, danh sách nạp lại và chọn mục đầu — mất mục cũ của giao dịch.

QSao logo màu xanh/đỏ mà file vector chỉ vẽ màu đen?›

Tô lúc chạy bằng ImageView.setColorFilter(màu), màu lấy theo kiểu của mục qua Icons.colorOf. Nhờ vậy một file logo dùng cho cả thu lẫn chi, không phải vẽ hai bản.

QSao CSDL lưu tên logo chứ không lưu R.drawable.xxx?›

R.drawable.xxx chỉ là một số nguyên do aapt2 đánh lúc build, thêm/bớt file là số có thể đổi. Lưu số đó thì lần build sau logo trong CSDL trỏ sai hình. Lưu tên như "anuong" thì luôn đổi đúng qua Icons.resOf.

QSao vào màn Sửa mục bằng bấm giữ, không bấm thường?›

Bấm thường đã là mở danh sách chọn. Spinner là AdapterView, gọi setOnClickListener sẽ ném lỗi; setOnItemClickListener cũng không dùng được với Spinner. Bấm giữ (setOnLongClickListener) thì được, đã thử trên máy ảo.

D · Luồng và vòng đời

QGửi Transaction qua Intent thì màn Sửa cầm cùng đối tượng với Home à?›

Không. Serializable qua Intent là bản sao (được đóng gói gửi qua hệ thống rồi dựng lại). Sửa ở EditAct không đụng tới danh sách của Home; Home thấy thay đổi là nhờ onResume() đọc lại CSDL.

QThêm mục bằng dấu + xong quay về, sao danh sách mục có mục mới?›

AddAct.onResume() gọi reloadCategories(id mục đang chọn): đọc lại cây từ CSDL, rồi chọn lại đúng mục cũ nếu nó còn. Mục mới có trong danh sách nhưng không tự được chọn — muốn vậy xem K11.

QBấm dấu + cạnh "mục cha" trong màn Thêm mục thì chuyện gì xảy ra?›

Mở thêm một AddCategoryAct nữa chồng lên, cùng kiểu thu/chi. Thêm xong nó finish(), màn dưới chạy onResume() → reloadParents() nên mục vừa thêm có ngay trong danh sách mục cha.

E · Câu hỏi mở

QMuốn có nhiều ví (ví tiền mặt, tài khoản ngân hàng) thì sửa CSDL thế nào?›

Thêm bảng tblWallet(id, name, note) và cột idWallet trong tblTransaction. Thêm lớp Wallet, Transaction có thêm wallet : Wallet. Home thêm Spinner chọn ví, getTransactions thêm điều kiện idWallet = ?. Tăng DB_VERSION.

QTính số dư hiện tại của ví (không chỉ trong ngày)?›

Tổng mọi khoản thu trừ tổng mọi khoản chi, không lọc ngày: SELECT SUM(CASE WHEN c.idType = 1 THEN t.amount ELSE -t.amount END) FROM tblTransaction t JOIN tblCategory c ON c.id = t.idCategory.

QDùng Room thì bỏ được gì?›

Bỏ DBHelper, valuesOf, categoryOf và các vòng đọc Cursor; Room tự sinh. Nhưng phần nối cây mục (loadCategories) và thuật toán xếp theo tần suất vẫn phải tự viết, vì đó là logic nghiệp vụ, không phải đọc/ghi.
